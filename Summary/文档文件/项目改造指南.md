### 2.1 新增映射加载函数

在`extractor.py`开头添加：

```python
import json
import os

def load_mapping_detail(mapping_file):
    """
    加载mapping_detail.json
    返回: {section_id: {contributing_sources: [...]}}
    """
    try:
        with open(mapping_file, 'r', encoding='utf-8') as f:
            mapping = json.load(f)
        return mapping
    except FileNotFoundError:
        print(f"⚠️ 警告：映射文件未找到 {mapping_file}，使用全量源文件")
        return None

def get_relevant_sources_for_section(section_id, mapping_detail):
    """
    根据mapping_detail.json，查询某小类应该从哪些源文件提取
    
    参数:
        section_id: 小类ID (1-5)
        mapping_detail: 映射字典
    
    返回:
        [(file_name, weight, relevant_sections), ...]
    """
    relevant_sources = []
    
    if not mapping_detail or 'sections' not in mapping_detail:
        return relevant_sources
    
    sections = mapping_detail['sections']
    
    # 查找对应section_id
    for section_key, section_data in sections.items():
        if section_data.get('id') == section_id:
            sources = section_data.get('contributing_sources', [])
            for source in sources:
                relevant_sources.append({
                    'file': source.get('file'),
                    'weight': source.get('weight', 1.0),
                    'relevant_sections': source.get('relevant_sections', [])
                })
            break
    
    return relevant_sources
```

### 2.2 改造extract_all_sections函数

**在extract_all_sections中集成映射逻辑**:

```python
def extract_all_sections(config, mapping_detail=None):
    """
    改造版：根据映射关系，逐小类提取内容
    
    参数:
        config: config.py中的配置
        mapping_detail: mapping_detail.json的内容
    """
    results = {}
    
    for section in config.SECTIONS:
        section_id = section["id"]
        section_key = section["key"]
        
        print(f"\n【阶段一：提取】 {section['title']} ...")
        
        # 第1步：获取该小类对应的源文件列表
        if mapping_detail:
            relevant_sources = get_relevant_sources_for_section(section_id, mapping_detail)
            print(f"  📋 关联源文件数：{len(relevant_sources)}")
        else:
            # 降级：没有映射文件则使用全量源文件
            relevant_sources = [{'file': f, 'weight': 1.0} for f in get_all_source_files()]
            print(f"  ⚠️  未找到映射，使用全量源文件({len(relevant_sources)}个)")
        
        # 第2步：读取和融合相关源文件的内容
        combined_content = ""
        for source in relevant_sources:
            file_path = os.path.join(config.INPUT_DIR, source['file'])
            if not os.path.exists(file_path):
                print(f"    ⚠️ 文件不存在：{source['file']}")
                continue
            
            try:
                with open(file_path, 'r', encoding='utf-8') as f:
                    content = f.read()
                weight = source.get('weight', 1.0)
                # 为内容添加元数据标注，便于LLM理解来源权重
                combined_content += f"\n\\n【来自{source['file']}（权重{weight}）】\\n{content}"
            except Exception as e:
                print(f"    ❌ 读取失败：{e}")
        
        # 第3步：提取（调用LLM）
        try:
            extracted = extract_section_with_llm(
                title=section['title'],
                description=section['description'],
                combined_content=combined_content,
                config=config
            )
            results[section_key] = extracted
            print(f"  ✓ 提取完成，获得 {len(extracted)} 条记录")
        except Exception as e:
            print(f"  ❌ 提取失败：{e}")
            results[section_key] = []
    
    return results
```

## 🚨 常见问题与解决

### Q1: mapping_detail.json 找不到
**A**: 确保文件在config.py指定的路径中
```python
MAPPING_DETAIL_FILE = "mapping_detail.json"  # 确保位置正确
```

### Q2: 提取出来的内容跨越了多个小类
**A**: 检查config.py中各section的exclude字段是否明确足
```python
"exclude": "其他8个建设小类及发展基础、发展环境等内容"
```

### Q3: 某些源文件权重很低(< 10%)，应该忽略吗？
**A**: 不应该忽略，这些是补充内容，会增加多角度视角

### Q4: 改造后字数突然变少或变多
**A**: 微调word_range参数
```python
"word_range": [2000, 3000]  # 根据实际输出调整
```
